### MÃ VẼ USE CASE (PLANTUML) - TOÀN DIỆN

```plantuml
@startuml
left to right direction
skinparam packageStyle rectangle

actor "Khách hàng" as KH
actor "Admin" as AD

rectangle "Hệ thống quản lý website bán sách" {
    
    usecase "Đăng nhập" as Login
    
    ' Use Cases Khách hàng
    usecase "Xem danh mục & Sách" as KH_UC1
    usecase "Quản lý giỏ hàng" as KH_UC2
    usecase "Đặt hàng & Thanh toán" as KH_UC3
    usecase "Theo dõi đơn hàng" as KH_UC4
    usecase "Quản lý hồ sơ & Yêu thích" as KH_UC5
    usecase "Xem thông báo" as KH_UC6
    usecase "Chat hỗ trợ" as KH_UC7

    ' Use Cases Admin
    usecase "Quản lý Sản phẩm & Danh mục" as AD_UC1
    usecase "Quản lý Tác giả & NXB" as AD_UC2
    usecase "Quản lý Đơn hàng" as AD_UC3
    usecase "Quản lý Nhập hàng & NCC" as AD_UC4

    usecase "Báo cáo doanh thu" as AD_UC7

    ' Kết nối Khách hàng
    KH -- KH_UC1

    KH -- KH_UC4
    KH -- KH_UC5
    KH -- KH_UC6
    KH -- KH_UC7

    AD_UC1 -- AD
    AD_UC2 -- AD
    AD_UC3 -- AD
    AD_UC4 -- AD
    AD_UC5 -- AD
    AD_UC6 -- AD
    AD_UC7 -- AD
    AD_UC8 -- AD

    ' Includes Admin
```
