### MÃ VẼ USE CASE (PLANTUML)

```plantuml
@startuml
left to right direction
skinparam packageStyle rectangle

actor "Khách hàng" as KH
actor "Admin" as AD
actor "Hệ thống" as System

rectangle "Hệ thống Bookstore" {
    
    usecase "Đăng nhập/Đăng ký" as Login
    
    ' Nhóm Khách hàng
    package "Nhóm Khách hàng" {
        usecase "Xem & Tìm kiếm Sách" as KH_UC1
        usecase "Quản lý Giỏ hàng" as KH_UC2
        usecase "Thanh toán (VNPay/COD)" as KH_UC3
        usecase "Theo dõi Đơn hàng" as KH_UC4
        usecase "Quản lý Yêu thích" as KH_UC5
        usecase "Chat AI Hỗ trợ" as KH_UC6
    }

    ' Nhóm Admin
    package "Nhóm Quản trị" {
        usecase "Quản lý Kho & Sản phẩm" as AD_UC1
        usecase "Xử lý Đơn hàng" as AD_UC2
        usecase "Quản lý Nhập hàng" as AD_UC3
        usecase "Thống kê Doanh thu" as AD_UC4
        usecase "Quản lý Khách hàng & Tài khoản" as AD_UC5
        usecase "Chat hỗ trợ trực tiếp" as AD_UC6
    }

    ' Nhóm Hệ thống
    usecase "Xử lý Thanh toán tự động (IPN)" as SYS_UC1

    ' Kết nối
    KH -- KH_UC1
    KH -- KH_UC2
    KH -- KH_UC3
    KH -- KH_UC4
    KH -- KH_UC5
    KH -- KH_UC6

    KH_UC2 ..> Login : <<include>>
    KH_UC3 ..> Login : <<include>>
    KH_UC4 ..> Login : <<include>>

    AD -- AD_UC1
    AD -- AD_UC2
    AD -- AD_UC3
    AD -- AD_UC4
    AD -- AD_UC5
    AD -- AD_UC6

    AD_UC1 ..> Login : <<include>>
    AD_UC2 ..> Login : <<include>>
    AD_UC3 ..> Login : <<include>>

    System -- SYS_UC1
}
@enduml
```
